
\begin{frame}{Appendix: GPR-kernels}

  \begin{itemize}
    \itemsep 1em
    \item Radial basis function (RBF)
    \[ k_\mathrm{RBF}(\mathbf{x}, \mathbf{x}') = \exp{\left(-\frac{||\mathbf{x}-\mathbf{x}'||^2}{2\sigma^2}\right)} \]
    % \item Mat\'ern
    % \[ k^\nu_\mathrm{Matern}(\mathbf{x}, \mathbf{x}') = \sigma^2 \frac{2^{1-\nu}}{\Gamma(\nu)} \left(\sqrt{2\nu}\frac{\sqrt{||\mathbf{x}-\mathbf{x}'||^2}}{\rho}\right)^\nu K_\nu\left(\sqrt{2\nu}\frac{\sqrt{||\mathbf{x}-\mathbf{x}'||^2}}{\rho}\right) \]
    \item Periodic
    \[ k_\mathrm{periodic}(\mathbf{x}, \mathbf{x}') = \exp{-\frac{2 \sin^2{\left(\pi \sqrt{||\mathbf{x}-\mathbf{x}'||^2} / p\right)}}{l^2}} \]
    \item RBF-periodic (RBFP)
    \[ k_\mathrm{RBFP}(\mathbf{x}, \mathbf{x}') = k_\mathrm{RBF}(\mathbf{x}, \mathbf{x}') \times k_\mathrm{periodic}(\mathbf{x}, \mathbf{x}') \]
    \item Stacked-RBFP
    \[ k^N_\mathrm{Stacked-RBFP}(\mathbf{x}, \mathbf{x}') = \sum_{i=0}^{N} {k_\mathrm{RBFP}^i(\mathbf{x}, \mathbf{x}')} \]
  \end{itemize}
  
\end{frame}
