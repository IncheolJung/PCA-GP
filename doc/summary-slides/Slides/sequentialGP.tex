\begin{frame} {PCA-GP}

  \begin{columns}

    \column{0.5\linewidth}
    \begin{figure}
      % \includegraphics[width=\linewidth]{example-image}
      \includegraphics[width=.85\linewidth]{PCA-GP/fig.pdf}
      \caption{PCA to compress prediction target fo GPR}
    \end{figure}

    \column{0.5\linewidth}
    \begin{itemize}
      \item Data of interest is 2D: [frequency, input angle].
      \item Direct prediction of the dataset is expensive and ineffective.
      \item PCA is employed to reduce the dataset, where the GPR model predicts the coefficient $\mathbf{Z'}$.
      \item This \textbf{Reduced Basis} approach is also incorporated with \textbf{VMAS}, thus minimizing computational load while maintaining accuracy.
    \end{itemize}
  
  \end{columns}
  
\end{frame}
